    Models¶

    class Model(BaseModel):
    """Base class of the models.

    A subclass with a `Meta` is a model whose data can be read and written. A
    subclass without one is abstract (a base for other models).
    """

    objects: ClassVar[ManagerDescriptor] = ManagerDescriptor()
    """The manager that reads and writes the model's data. Class access only.

    `objects` is reserved: a subclass can only replace it with a
    :class:`ModelManager` (see there to add methods of your own).
    """

    nan_as_none: ClassVar[bool] = True
    """Store a `NaN` given to an `X | None` field as `None` (default `True`)."""

    __smallcat_meta__: ClassVar[ModelMeta | None] = None
    __smallcat_schema__: ClassVar[ModelSchema | None] = None

    @classmethod
    def __pydantic_init_subclass__(cls, **kwargs: Any) -> None:  # noqa: ANN401
        """Validate `Meta` and derive the schema when a model is defined."""
        super().__pydantic_init_subclass__(**kwargs)
        declared = cls.__dict__.get("objects", None)
        if "objects" in cls.model_fields or (
            declared is not None
            and not isinstance(declared, ModelManager | ManagerDescriptor)
        ):
            msg = (
                f"'objects' is a reserved name on models: in {cls.__name__} it "
                "must be a ModelManager instance (e.g. objects = "
                f"CustomerManager()), got {type(declared).__name__}"
            )
            raise ModelDefinitionError(msg)
        if "save" in cls.model_fields:
            msg = (
                f"'save' is a reserved name on models: {cls.__name__} cannot "
                "have a field called 'save' (it is the method that writes the "
                "instance)"
            )
            raise ModelDefinitionError(msg)
        meta_class = getattr(cls, "Meta", None)
        if meta_class is None:
            return
        meta = parse_meta(cls.__name__, meta_class)
        schema = _build_schema(cls)
        problems = unsupported_fields(
            schema,
            meta.file_format,
            writable=bool(meta.allowed_write_modes),
        )
        if problems:
            details = "\n  - ".join(problems)
            msg = (
                f"{cls.__name__} has fields its file format cannot hold:\n  - {details}"
            )
            raise ModelDefinitionError(msg)
        cls.__smallcat_meta__ = meta
        cls.__smallcat_schema__ = schema

    def save(
        self,
        *,
        mode: ModelWriteMode | str = ModelWriteMode.MERGE,
        merge_keys: Sequence[str] | None = None,
    ) -> None:
        """Validate this instance and write it as one row.

        Shortcut for ``type(self).objects.save(self, mode=..., merge_keys=...)``.
        By default it merges: the row is inserted if its key is new and updates
        the row with the same key otherwise, by ``Meta.merge_keys`` (or
        ``merge_keys``). Every column is written, so a field left at its default
        overwrites the stored value.

        Only Delta tables can merge: on any other format the default mode
        raises :class:`ModelUnsupportedWriteModeError` and nothing is written.
        Pass ``mode="overwrite"`` explicitly to replace the whole dataset with
        this row.

        Each call is one write (one Delta commit). To save many rows use
        ``Model.objects.save([...], mode=...)``.

        Args:
            mode: A :class:`ModelWriteMode` or its value. Default ``merge``.
            merge_keys: Fields that identify the row, with ``mode="merge"``.
                By default ``Meta.merge_keys``.

        Raises:
            ValueError: If `mode` is not a :class:`ModelWriteMode`.
            ModelNotRegisteredError: If the model is not registered.
            ModelQueryError: If ``merge`` has no keys (neither in the call nor in
                ``Meta``), a key is not a field of the model, or `merge_keys`
                is given with a mode other than ``merge``.
            ModelWriteNotAllowedError: If the model does not allow the mode.
            ModelUnsupportedWriteModeError: If the file format cannot write
                with the mode (``merge`` on CSV, Parquet or Excel).
            ModelSchemaValidationError: If the row does not match the model.
        """
        type(self).objects.save(self, mode=mode, merge_keys=merge_keys)

    @model_validator(mode="before")
    @classmethod
    def _nan_to_none(cls, data: object) -> object:
        """Turn `NaN` into `None` in the fields that admit `None`.

        Empty cells come from pandas as `NaN`. Fields that do not admit `None`
        keep their value: `NaN` is a valid `float`.
        """
        if not cls.nan_as_none or not isinstance(data, dict):
            return data
        fixed: dict[Any, Any] | None = None
        for name, info in cls.model_fields.items():
            value = data.get(name)
            if (
                isinstance(value, float)
                and math.isnan(value)
                and _admits_none(info.annotation)
            ):
                if fixed is None:
                    fixed = dict(data)
                fixed[name] = None
        return data if fixed is None else fixed

    class ModelRegistry:
    """Registers models against loaded connections."""

    def __init__(self, connections: Connections | None = None) -> None:
        """Create a registry.

        Args:
            connections: Connections to start with. More can be added with
                :meth:`load_connections`.
        """
        self._connections = Connections()
        if connections is not None:
            self._connections.update(connections)
        self._registrations: dict[type, Registration] = {}

    @property
    def connections(self) -> Connections:
        """The connections loaded in this registry."""
        return self._connections

    def load_connections(self, connections: Connections) -> None:
        """Add connections that models can then be registered against.

        Args:
            connections: The connections, already validated by their loader.

        Raises:
            DuplicateConnectionError: If a name is already loaded.
        """
        self._connections.update(connections)

    def register(
        self,
        model: "type[Model]",
        *,
        connection: str,
        allowed_write_modes: Iterable[ModelWriteMode | str] | None = None,
        replace: bool = False,
    ) -> Registration:
        """Register a model against a loaded connection.

        Args:
            model: The model class.
            connection: Name of a connection already loaded in this registry.
            allowed_write_modes: Restricts the write modes for this
                registration. It can only narrow what the model's `Meta`
                allows; `[]` makes it read-only here. By default the model's
                own modes apply.
            replace: Allow replacing an existing registration of the model.

        Returns:
            The registration.

        Raises:
            ModelRegistrationError: If the model has no `Meta`, is already
                registered, `allowed_write_modes` asks for a mode the model
                does not allow, or its `Meta.merge_keys` / `update_columns` are
                not valid fields of the model (a key must also be required).
            ConnectionNotLoadedError: If the connection is not loaded.
        """
        meta: ModelMeta | None = getattr(model, "__smallcat_meta__", None)
        if meta is None:
            msg = f"{model.__name__} has no Meta: it cannot be registered"
            raise ModelRegistrationError(msg)
        self._connections.require(connection)
        _check_merge_fields(model, meta)
        if model in self._registrations and not replace:
            msg = (
                f"{model.__name__} is already registered "
                "(pass replace=True to replace it)"
            )
            raise ModelRegistrationError(msg)

        allowed = meta.allowed_write_modes
        if allowed_write_modes is not None:
            requested = frozenset(
                parse_enum(ModelWriteMode, m, "write mode") for m in allowed_write_modes
            )
            widened = sorted(m.value for m in requested - meta.allowed_write_modes)
            if widened:
                msg = (
                    f"{model.__name__}.Meta does not allow {widened}: registering "
                    "can only restrict write modes, not add them"
                )
                raise ModelRegistrationError(msg)
            allowed = requested

        registration = Registration(model, connection, allowed)
        self._registrations[model] = registration
        return registration

    def unregister(self, model: "type[Model]") -> None:
        """Remove a model's registration, if it has one."""
        self._registrations.pop(model, None)

    def get(self, model: "type[Model]") -> Registration:
        """Return a model's registration.

        Raises:
            ModelNotRegisteredError: If the model is not registered here.
        """
        try:
            return self._registrations[model]
        except KeyError:
            msg = (
                f"{model.__name__} is not registered. Register it first: "
                f"registry.register({model.__name__}, connection='<name>')"
            )
            raise ModelNotRegisteredError(msg) from None

    def entry_for(
        self,
        model: "type[Model]",
        write_mode: ModelWriteMode | None = None,
        merge_keys: Sequence[str] | None = None,
    ) -> Any:  # noqa: ANN401
        """Build the catalog entry that reads or writes a registered model.

        Args:
            model: The model class.
            write_mode: The mode of the write about to happen. Delta tables take
                it as their `mode`; other formats need nothing.
            merge_keys: The keys of a `merge` write. They replace
                `Meta.merge_keys`; with none given, those are used.

        Returns:
            A catalog entry bound to this registry's connections.
        """
        registration = self.get(model)
        meta: ModelMeta = model.__smallcat_meta__  # type: ignore[assignment]
        save_options = None
        if meta.file_format is ModelFileFormat.DELTA_TABLE:
            # `update_columns` only means something with mode `merge`, so it
            # stays out of reads and of every other write.
            save_options = dict(meta.save_options or {})
            if write_mode is ModelWriteMode.MERGE:
                keys = merge_keys if merge_keys is not None else meta.merge_keys
                save_options["mode"] = write_mode.value
                save_options["merge_keys"] = list(keys or [])
            else:
                save_options.pop("update_columns", None)
                if write_mode is not None:
                    save_options["mode"] = write_mode.value
        entry = build_entry(meta, registration.connection, save_options)
        entry.bind_connections(self._connections)
        return entry

    def __contains__(self, model: object) -> bool:
        """Return whether a model is registered here."""
        return model in self._registrations

    def __iter__(self) -> Iterator["type[Model]"]:
        """Iterate over the registered models."""
        return iter(self._registrations)

    def __len__(self) -> int:
        """Return the number of registered models."""
        return len(self._registrations)

        model: The model class.
        connection: Name of a connection already loaded in this registry.
        allowed_write_modes: Restricts the write modes for this
            registration. It can only narrow what the model's `Meta`
            allows; `[]` makes it read-only here. By default the model's

            not valid fields of the model (a key must also be required).
        ConnectionNotLoadedError: If the connection is not loaded.
    """

    if meta is None:
        msg = f"{model.__name__} has no Meta: it cannot be registered"
        raise ModelRegistrationError(msg)
    self._connections.require(connection)
    _check_merge_fields(model, meta)
    if model in self._registrations and not replace:
        msg = (
            f"{model.__name__} is already registered "
            "(pass replace=True to replace it)"
        )
        raise ModelRegistrationError(msg)

    if allowed_write_modes is not None:
        requested = frozenset(
            parse_enum(ModelWriteMode, m, "write mode") for m in allowed_write_modes

                "can only restrict write modes, not add them"

                f"A manager must be declared as 'objects', not '{name}' "
                f"(in {owner.__name__})"

        Reading a model whose table or file has not been written yet fails.
        This creates it with no rows and the model's schema (a Delta table, or
        a CSV, Parquet or Excel file). Nothing is done if it already exists.

        A model that allows no write mode (read-only) is not created: a
        warning is emitted instead of an error.

        Returns:
            True if it was created; False if it already existed or the model
            is read-only.

            msg = f"{model.__name__} is read-only: its dataset was not created"

        entry.create_empty(schema)

        merge_keys: Sequence[str] | None = None,

        The mode is required because it decides what happens to the data that
        already exists:

        * ``overwrite`` replaces the dataset entirely.
        * ``append`` adds the rows (only Delta tables can).
        * ``merge`` inserts the rows with a new key and updates the ones whose
          key exists, by ``merge_keys`` (only Delta tables can). It never
          deletes rows. The keys of the call replace those of ``Meta``.

        Args:
            data: A DataFrame, an instance of the model or a list of them.
            mode: A :class:`ModelWriteMode` or its value.

                By default ``Meta.merge_keys``.

        Raises:
            ValueError: If `mode` is not a :class:`ModelWriteMode`.
            ModelNotRegisteredError: If the model is not registered.
            ModelQueryError: If `merge_keys` is given with a mode other than
                ``merge``, if ``merge`` has no keys (neither in the call nor in
                ``Meta``) or a key is not a field of the model.
            ModelWriteNotAllowedError: If the model (or its registration) does
                not allow this mode; read-only models allow none.
            ModelUnsupportedWriteModeError: If the file format cannot write
                with this mode.
            ModelSchemaValidationError: If the data does not match the model.
            TypeError: If `data` is not a DataFrame or model instances.
        """
        write_mode = parse_enum(ModelWriteMode, mode, "write mode")
        model = self.model
        registry = current_registry()
        registration = registry.get(model)
        meta: ModelMeta = model.__smallcat_meta__  # type: ignore[assignment]
        spec: ModelSchema = model.__smallcat_schema__  # type: ignore[assignment]

        allowed = registration.allowed_write_modes
        if write_mode not in allowed:
            if not allowed:
                msg = f"{model.__name__} is read-only: it allows no write mode"
            else:
                msg = (
                    f"{model.__name__} does not allow '{write_mode.value}' "
                    f"(allowed: {sorted(m.value for m in allowed)})"
                )
            raise ModelWriteNotAllowedError(msg)
        reason = unsupported_reason(meta, write_mode)
        if reason:
            msg = (
                f"{model.__name__} cannot be saved with '{write_mode.value}': {reason}"
            )
            raise ModelUnsupportedWriteModeError(msg)

        keys = _merge_keys_for(model, meta, spec, write_mode, merge_keys)
        frame = validate_frame(_to_frame(model, data), spec, meta.validation_mode)
        registry.entry_for(model, write_mode, keys).save_polars(frame)

    Attributes:
        model: The model class.
        filters: Polars predicates, combined with AND.
        where_sql: An SQL predicate (DuckDB), alternative to `filters`.
        columns: The columns to read, or `None` for all of them.

    filters: tuple[pl.Expr, ...] = ()
    where_sql: str | None = None
    columns: tuple[str, ...] | None = None

    # ---------- Building ----------

    def filter(self, *predicates: pl.Expr) -> Self:
        """Keep the rows that satisfy every Polars predicate.

        Args:
            *predicates: Polars expressions, e.g. `pl.col("id") > 10`.

        Returns:
            A new query.

        Raises:
            ModelQueryError: If no predicate is given, one is not a Polars
                expression, or the query already uses `where()`.
        """
        if not predicates:
            msg = "filter() needs at least one Polars expression"
            raise ModelQueryError(msg)
        for predicate in predicates:
            if not isinstance(predicate, pl.Expr):
                msg = (
                    "filter() takes Polars expressions such as pl.col('id') > 10, "
                    f"not {type(predicate).__name__}; use where() for an SQL string"
                )
                raise ModelQueryError(msg)
        if self.where_sql is not None:
            raise ModelQueryError(_MIXED)
        return replace(self, filters=(*self.filters, *predicates))

    def where(self, sql: str | None = None, **lookups: Any) -> Self:  # noqa: ANN401
        """Keep the rows that satisfy an SQL predicate and/or lookups.

        Lookups are Django-style: ``field=value`` or ``field__operator=value``
        (see :mod:`smallcat.models.lookups` for the operators). They are checked
        against the model's fields and types right here. Several conditions are
        joined with AND, and so are `sql` and the lookups when both are given.

        Args:
            sql: An SQL predicate (DuckDB), e.g. `"id > 10 OR name = 'a'"`. It
                can also be passed as the first positional argument. To filter
                a field that is itself called `sql`, write `sql__exact=...`.
            **lookups: The lookups, e.g. `job_id__lte=2, country="ES"`.

        Returns:
            A new query.

        Raises:
            ModelQueryError: If there is neither an SQL predicate nor lookups,
                the predicate is empty, a lookup is not valid for the model,
                `where()` was already used, or the query already uses
                `filter()`.
        """
        if sql is None and not lookups:
            msg = "where() takes an SQL predicate (sql='...') or lookups (id__gt=1)"
            raise ModelQueryError(msg)
        if sql is not None and (not isinstance(sql, str) or not sql.strip()):
            msg = "where() takes a non-empty SQL predicate string"
            raise ModelQueryError(msg)
        if self.filters:
            raise ModelQueryError(_MIXED)
        if self.where_sql is not None:
            msg = (
                "where() was already used: put every condition in one call "
                "(joined with AND)"
            )
            raise ModelQueryError(msg)
        if not lookups:
            return replace(self, where_sql=sql)
        spec: ModelSchema = self.model.__smallcat_schema__  # type: ignore[assignment]
        rendered = lookups_to_sql(spec, lookups)
        combined = rendered if sql is None else f"({sql}) AND ({rendered})"
        return replace(self, where_sql=combined)

    def select(self, *names: str) -> Self:
        """Read only some of the model's fields.

        Args:
            *names: Field names.

        Returns:
            A new query.

        Raises:
            ModelQueryError: If a name is not a field of the model, is repeated,
                or `select()` was already used.
        """
        if not names:
            msg = "select() needs at least one field name"
            raise ModelQueryError(msg)
        if self.columns is not None:
            msg = "select() was already used"
            raise ModelQueryError(msg)
        unknown = [n for n in names if n not in self.model.model_fields]
        if unknown:
            msg = (
                f"select(): {unknown} are not fields of {self.model.__name__} "
                f"(fields: {list(self.model.model_fields)})"
            )
            raise ModelQueryError(msg)
        if len(set(names)) != len(names):
            msg = "select(): a field is repeated"
            raise ModelQueryError(msg)
        return replace(self, columns=tuple(names))

    # ---------- Running ----------

    def lazy(self) -> pl.LazyFrame:
        """Return the query as a Polars LazyFrame.

        Parquet and Delta are scanned natively, so filters and columns are
        pushed down to the storage. Other formats are read in full. A query
        with `where()` runs its SQL now and wraps the result.

        Only the schema (columns and types) is checked here, since no data has
        been read. Two checks need the data and are *not* made on a frame you
        collect yourself: nulls in fields that are not optional (use `all()`
        for that), and in `COERCE` mode a value that cannot be cast, which then
        fails with a Polars error when the frame is collected.

        Returns:
            The LazyFrame, validated against the model's schema.
        """
        if self.where_sql is not None:
            return self.all().lazy()
        return self._plan(_context(self.model))[0]

    def all(self, *, as_models: bool = False) -> "pl.DataFrame | list[M]":
        """Run the query and return a DataFrame, or model instances.

        Args:
            as_models: If true, return one instance of the model per row
                instead of a DataFrame. Each row is validated by Pydantic, so
                field constraints apply.

        Returns:
            The rows as a DataFrame with the model's columns in its order, or
            as a list of instances when `as_models` is true.

        Raises:
            ModelQueryError: If `as_models` is true and the query uses
                `select()`, since instances need every required field.
            ModelSchemaValidationError: If the data does not match the model.
        """
        if not as_models:
            return self._frame()
        if self.columns is not None:
            msg = "as_models() needs every field: remove select()"
            raise ModelQueryError(msg)
        rows = self._frame().iter_rows(named=True)
        return [self.model.model_validate(row) for row in rows]

    def _frame(self) -> pl.DataFrame:
        """Run the query and return the validated DataFrame."""
        ctx = _context(self.model)
        selected = list(self.columns) if self.columns else None
        if self.where_sql is not None:
            df = ctx.entry.load_polars(where=self.where_sql, columns=selected)
            return validate_frame(df, ctx.spec, ctx.mode, selected)

        plan, raw = self._plan(ctx)
        try:
            df = plan.collect()
        except (pl.exceptions.InvalidOperationError, pl.exceptions.ComputeError):
            if ctx.mode is ModelValidationMode.COERCE:
                # Find out which column and how many values, from the raw data.
                raw_df = (raw.select(selected) if selected else raw).collect()
                validate_frame(raw_df, ctx.spec, ctx.mode, selected)
            raise
        check_nulls(df, ctx.spec, selected)
        return df

    def as_models(self) -> "list[M]":
        """Run the query and return instances of the model.

        Shortcut for ``all(as_models=True)``.

        Returns:
            One instance per row.

        Raises:
            ModelQueryError: If the query uses `select()`, since instances
                need every required field.
        """
        return self.all(as_models=True)

    def _plan(self, ctx: _Context) -> tuple[pl.LazyFrame, pl.LazyFrame]:
        """Build the lazy plan and return it with the unfiltered scan."""
        selected = list(self.columns) if self.columns else None
        raw = ctx.entry.scan_polars()
        prepared = prepare_lazy(raw, ctx.spec, ctx.mode, selected)
        frame = prepared.frame
        if self.filters:
            frame = frame.filter(*self.filters)
        return frame.select(prepared.select), raw

    smallcat.models.lookups ¶

    Django-style lookups for where(): job_id__lte=2 and friends.

    A lookup is written as <field> or <field>__<operator> and a value:

    Customer.objects.where(job_id=42)
Customer.objects.where(job_id__lte=2, country="ES")
Customer.objects.where(status__in=["new", "open"], deleted_at__isnull=True)

    Lookups are resolved when the query is built, against the fields of the model: an unknown field, an operator the field's type does not offer or a value of the wrong type raises :class:ModelQueryError right where you write it. Nothing is defined per model: the fields and their types are read from the model itself.

    They are rendered to an SQL predicate (DuckDB). Names come from the model and values are typed literals that this module quotes, so nothing the caller passes is ever interpreted as SQL.

    Operators:

    • exact: equal. It is the default when no operator is given. Takes one value, or None (the same as isnull=True).
    • ne: not equal. One value, or None.
    • lt, lte, gt, gte: less than, less than or equal, greater than, greater than or equal. One value.
    • in: equal to any of the values. A list, tuple or set; an empty one matches nothing.
    • range: between two values, both included. A pair (low, high).
    • isnull: True for is null, False for is not null. A bool.
    • contains, startswith, endswith: text matches. A str; % and _ in it mean themselves.

    Which operators a field accepts depends on its type (see :data:OPERATORS).

    OPERATORS module-attribute ¶

    OPERATORS: dict[str, tuple[str, ...]] = {'int': _ORDERED, 'float': _ORDERED, 'date': _ORDERED, 'datetime': _ORDERED, 'time': _ORDERED, 'duration': _ORDERED, 'str': _TEXT, 'bool': ('exact', 'ne', 'isnull'), 'other': ('isnull',)}

    The operators each kind of field accepts. other is lists and bytes.
